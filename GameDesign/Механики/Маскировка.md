```dataviewjs
const { fieldModifier: f } = app.plugins.plugins["metadata-menu"].api;

const currentFile = dv.current().file;

const children = dv.pages()
    .where(p => 
        p.Механики &&
        p.Механики.some(m => m.path === currentFile.path)
    );

const total = children.length;

const done = children
    .where(p => p["Реализовано"] === true)
    .length;

const percent = total > 0 ? Math.round((done / total) * 100) : 0;

// Записываем число в frontmatter
await app.fileManager.processFrontMatter(
    app.vault.getAbstractFileByPath(currentFile.path),
    (frontmatter) => {
        frontmatter["Статус"] = percent;
    }
);

dv.table(
    ["Название", "Реализовано"],
    await Promise.all(
        children
            .sort(p => p["Реализовано"])
            .map(async p => [
                p.file.link,
                await f(dv, p, "Реализовано")
            ])
    )
);
```
